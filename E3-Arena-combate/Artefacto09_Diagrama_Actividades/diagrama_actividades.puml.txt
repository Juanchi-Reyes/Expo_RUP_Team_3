
@startuml
title Diagrama de actividades - Combate RPG Manager

|Jugador|
start
:Seleccionar rival;
:Iniciar combate;

|Sistema|
:Preparar personajes;
:Asignar vida inicial;
:Definir primer turno;

while (¿Combate finalizado?) is (No)

    |Jugador|
    :Seleccionar acción;
    :Elegir ataque o habilidad;

    |Sistema|
    :Evaluar acción;

    if (¿El rival esquiva?) then (Sí)
        :Registrar ataque esquivado;
    else (No)
        :Aplicar daño al rival;
    endif

    :Actualizar puntos de vida;

    if (¿La vida del rival es 0?) then (Sí)
        :Declarar ganador;
        :Registrar resultado;
        break
    else (No)
        :Cambiar turno;
    endif

endwhile (Sí)

|Sistema|
:Mostrar resultado del combate;
stop

@enduml
